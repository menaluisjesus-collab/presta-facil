// Registrar Cobro y Enviar el Ticket exacto al portapapeles y WhatsApp sin textos adicionales window.cobrar = function(loanId, clientName, cedula, phone, amount, faltantes, totalCuotas, frequency) { if(faltantes <= 0) { alert('El préstamo ya se encuentra pagado por completo.'); return; } const now = new Date(); const dateStr = now.toLocaleDateString('es-DO'); const nuevasFaltantes = faltantes - 1; const saldoRestante = nuevasFaltantes * amount; const paymentData = { loanId: loanId, client: clientName, amount: amount, timestamp: now.getTime(), dateStr: dateStr, year: now.getFullYear(), month: now.getMonth() + 1 }; paymentsRef.push().set(paymentData, (error) => { if(!error) { loansRef.child(loanId).update({ faltantes: nuevasFaltantes }); const ticketStr = `============================== E&S PRESTA FÁCIL PRO RECIBO DE PAGO OFICIAL ============================== Fecha: ${dateStr} Cliente: ${clientName.toUpperCase()} Cédula: ${cedula} ------------------------------ Cuota Pagada: RD$ ${amount.toFixed(2)} Modalidad: ${frequency} Progreso: ${(totalCuotas - nuevasFaltantes)} de ${totalCuotas} cuotas Saldo Restante: RD$ ${saldoRestante.toFixed(2)} ------------------------------ ¡Gracias por mantener su crédito al día con nosotros!`; document.getElementById('modal-ticket-text').innerText = ticketStr; // Copiamos el ticket exacto al portapapeles navigator.clipboard.writeText(ticketStr).catch(err => { console.error("Error al copiar: ", err); }); // Enlace limpio directo solo con el formato de ticket para que elijas tu contacto document.getElementById('modal-whatsapp-btn').href = "https://wa.me/?text=" + encodeURIComponent(ticketStr); document.getElementById('modal-overlay').style.display = 'flex'; } else { alert('Error al registrar cobro: ' + error.message); } }); };
